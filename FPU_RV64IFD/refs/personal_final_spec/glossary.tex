% ------------------ Glossary Entries ------------------


\newglossaryentry{pipeline}{
    name={pipeline},
    description={A series of processing stages that execute instructions in overlapping steps}
}

\newglossaryentry{Harvard}{
    name={Harvard Architecture},
    description={A computer architecture with separate instruction and data memory paths}
}

\newglossaryentry{SoC}{
    name={System-on-Chip},
    description={An integrated circuit containing processing cores, memory, and peripherals}
}

\newglossaryentry{clockfreq}{
    name={Clock Frequency},
    description={The operating speed at which the processor core executes instructions}
